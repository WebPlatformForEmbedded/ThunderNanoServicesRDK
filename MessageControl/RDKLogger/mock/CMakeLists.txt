find_path(RDKLOGGER_INCLUDE_DIR rdk_logger.h)

if(NOT RDKLOGGER_INCLUDE_DIR)
    set(RDKLOGGER_INCLUDE_DIR "${CMAKE_CURRENT_BINARY_DIR}/include")
    file(MAKE_DIRECTORY "${RDKLOGGER_INCLUDE_DIR}")
    file(DOWNLOAD
        "https://raw.githubusercontent.com/rdkcentral/rdk_logger/5ccaad2e7fddb89941f377b2fdd917c5ddee0a21/include/rdk_logger.h"
        "${RDKLOGGER_INCLUDE_DIR}/rdk_logger.h"
        EXPECTED_HASH SHA256=f213c1e1659ae2d26fffd0f5ccb3ee71f1805099428c66710d712a2e0310b19a
        STATUS RDKLOGGER_DOWNLOAD_STATUS)
    list(GET RDKLOGGER_DOWNLOAD_STATUS 0 RDKLOGGER_DOWNLOAD_CODE)
    if(NOT RDKLOGGER_DOWNLOAD_CODE EQUAL 0)
        list(GET RDKLOGGER_DOWNLOAD_STATUS 1 RDKLOGGER_DOWNLOAD_ERROR)
        message(FATAL_ERROR "Could not download rdk_logger.h: ${RDKLOGGER_DOWNLOAD_ERROR}; set RDKLOGGER_INCLUDE_DIR for offline builds")
    endif()
endif()

add_library(${NAMESPACE}RDKLoggerMock SHARED RDKLoggerMock.c)

target_include_directories(${NAMESPACE}RDKLoggerMock
    PUBLIC $<BUILD_INTERFACE:${RDKLOGGER_INCLUDE_DIR}>)

install(TARGETS ${NAMESPACE}RDKLoggerMock
    EXPORT ${NAMESPACE}MessagingTargets
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} COMPONENT ${NAMESPACE}_Development
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} COMPONENT ${NAMESPACE}_Runtime
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} COMPONENT ${NAMESPACE}_Runtime)
